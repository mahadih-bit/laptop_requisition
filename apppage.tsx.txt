"use client";

import React, { useState, useEffect } from 'react';
import { useForm, FormProvider } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { RequisitionSchema, RequisitionFormData } from '@/lib/schema';
import { INITIAL_DATA } from '@/lib/constants';
import { 
  ClipboardCheck, 
  Cpu, 
  ShieldCheck, 
  Printer, 
  Save, 
  ChevronRight, 
  ChevronLeft,
  CheckCircle2
} from 'lucide-react';

// Sub-components
import PartAGeneral from '@/components/PartAGeneral';
import PartBWorkload from '@/components/PartBWorkload';
import PartCApproval from '@/components/PartCApproval';
import LivePreview from '@/components/LivePreview';

export default function RequisitionSystem() {
  const [step, setStep] = useState(1);
  const [isMISMode, setIsMISMode] = useState(false); // Toggle for demo purposes
  const [isSubmitted, setIsSubmitted] = useState(false);

  const methods = useForm<RequisitionFormData>({
    resolver: zodResolver(RequisitionSchema),
    defaultValues: INITIAL_DATA,
  });

  const onSubmit = (data: RequisitionFormData) => {
    console.log("Final Submission:", data);
    setIsSubmitted(true);
    // In production: await fetch('/api/requisition', { method: 'POST', body: JSON.stringify(data) })
  };

  const nextStep = () => setStep(s => Math.min(s + 1, 4));
  const prevStep = () => setStep(s => Math.max(s - 1, 1));

  if (isSubmitted) {
    return (
      <div className="min-h-screen flex flex-col items-center justify-center bg-slate-50 p-4">
        <div className="bg-white p-8 rounded-2xl shadow-xl max-w-md text-center">
          <CheckCircle2 className="w-16 h-16 text-green-500 mx-auto mb-4" />
          <h1 className="text-2xl font-bold text-slate-900">Requisition Submitted!</h1>
          <p className="text-slate-600 mt-2">Reference ID: REQ-{Math.floor(Math.random() * 10000)}</p>
          <p className="text-sm text-slate-500 mt-4">Your request has been forwarded to the Head of Department for initial approval.</p>
          <button 
            onClick={() => window.location.reload()}
            className="mt-6 px-6 py-2 bg-indigo-600 text-white rounded-lg hover:bg-indigo-700 transition"
          >
            Create New Requisition
          </button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-50 pb-20">
      {/* Header */}
      <header className="bg-white border-b sticky top-0 z-30">
        <div className="max-w-7xl mx-auto px-4 h-16 flex items-center justify-between">
          <div className="flex items-center gap-2">
            <div className="bg-indigo-600 p-2 rounded-lg">
              <Cpu className="text-white w-5 h-5" />
            </div>
            <h1 className="font-bold text-slate-800 hidden md:block">IT Assets Portal <span className="text-slate-400 font-normal">v3.0</span></h1>
          </div>
          
          <div className="flex items-center gap-4">
            <div className="flex items-center gap-2 bg-slate-100 p-1 rounded-full px-3">
              <div className="w-2 h-2 bg-green-500 rounded-full animate-pulse" />
              <span className="text-xs font-medium text-slate-600">User: Mahadi Hasan (30733)</span>
            </div>
            <button 
              onClick={() => setIsMISMode(!isMISMode)}
              className={`text-xs px-3 py-1 rounded-full border transition ${isMISMode ? 'bg-orange-100 border-orange-300 text-orange-700' : 'bg-white border-slate-300 text-slate-600'}`}
            >
              {isMISMode ? "Switch to User View" : "Switch to MIS View"}
            </button>
          </div>
        </div>
      </header>

      <main className="max-w-5xl mx-auto px-4 py-8">
        {/* Stepper */}
        <div className="mb-8 flex justify-between items-center relative">
          {[
            { id: 1, label: "Requester Details", icon: ClipboardCheck },
            { id: 2, label: "Workload Assessment", icon: Cpu },
            { id: 3, label: "Review & Preview", icon: ShieldCheck },
            { id: 4, label: "Approval Status", icon: CheckCircle2 },
          ].map((s) => (
            <div key={s.id} className="flex flex-col items-center z-10 flex-1">
              <div className={`w-10 h-10 rounded-full flex items-center justify-center transition ${step >= s.id ? 'bg-indigo-600 text-white' : 'bg-white text-slate-400 border border-slate-200'}`}>
                <s.icon className="w-5 h-5" />
              </div>
              <span className={`text-[10px] mt-2 font-semibold uppercase tracking-wider ${step >= s.id ? 'text-indigo-600' : 'text-slate-400'}`}>
                {s.label}
              </span>
            </div>
          ))}
          <div className="absolute top-5 left-0 w-full h-[2px] bg-slate-200 -z-0" />
          <div 
            className="absolute top-5 left-0 h-[2px] bg-indigo-600 transition-all duration-500 -z-0" 
            style={{ width: `${((step - 1) / 3) * 100}%` }}
          />
        </div>

        {/* Form Content */}
        <FormProvider {...methods}>
          <form onSubmit={methods.handleSubmit(onSubmit)} className="space-y-6">
            <div className="bg-white rounded-2xl shadow-sm border border-slate-200 p-6 md:p-8">
              {step === 1 && <PartAGeneral />}
              {step === 2 && <PartBWorkload isMISMode={isMISMode} />}
              {step === 3 && <LivePreview />}
              {step === 4 && <PartCApproval />}
            </div>

            {/* Navigation Buttons */}
            <div className="flex justify-between items-center">
              <button
                type="button"
                onClick={prevStep}
                disabled={step === 1}
                className="flex items-center gap-2 px-6 py-2.5 rounded-xl font-semibold text-slate-600 hover:bg-slate-100 disabled:opacity-0 transition"
              >
                <ChevronLeft className="w-4 h-4" /> Back
              </button>
              
              <div className="flex gap-3">
                <button
                  type="button"
                  className="hidden md:flex items-center gap-2 px-6 py-2.5 rounded-xl font-semibold text-slate-600 border border-slate-200 hover:bg-slate-50 transition"
                >
                  <Save className="w-4 h-4" /> Save Draft
                </button>
                
                {step < 4 ? (
                  <button
                    type="button"
                    onClick={nextStep}
                    className="flex items-center gap-2 px-8 py-2.5 rounded-xl font-semibold bg-indigo-600 text-white hover:bg-indigo-700 shadow-lg shadow-indigo-200 transition"
                  >
                    Continue <ChevronRight className="w-4 h-4" />
                  </button>
                ) : (
                  <button
                    type="submit"
                    className="flex items-center gap-2 px-8 py-2.5 rounded-xl font-semibold bg-green-600 text-white hover:bg-green-700 shadow-lg shadow-green-200 transition"
                  >
                    Final Submit <CheckCircle2 className="w-4 h-4" />
                  </button>
                )}
              </div>
            </div>
          </form>
        </FormProvider>
      </main>
    </div>
  );
}
